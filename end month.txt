End_Month =
SELECTCOLUMNS(
    DimMonth,
    "MonthStart", DATEVALUE(DimMonth[MonthStart]),
    "MonthEnd", EOMONTH(DATEVALUE(DimMonth[MonthStart]), 0),
    "MonthYear", FORMAT(
        DATEVALUE(DimMonth[MonthStart]),
        "MMM-yy"
    ),
    "MonthSort",
        YEAR(DATEVALUE(DimMonth[MonthStart])) * 100
        + MONTH(DATEVALUE(DimMonth[MonthStart]))
)


-------------------------------------------------------------------
Retention_Rate_Account =
VAR SelectedEndMonth =
    SELECTEDVALUE(End_Month[MonthStart])

VAR OpeningAccounts =
    CALCULATE(
        SUM(Sheet_1[opening_accounts_r12]),
        TREATAS(
            {SelectedEndMonth},
            Sheet_1[month_start]
        )
    )

VAR ExitedAccounts =
    CALCULATE(
        SUM(Sheet_1[exited_accounts_r12]),
        TREATAS(
            {SelectedEndMonth},
            Sheet_1[month_start]
        )
    )

RETURN
IF(
    ISBLANK(SelectedEndMonth)
        || OpeningAccounts <= 0,
    BLANK(),
    1 - DIVIDE(
        ExitedAccounts,
        OpeningAccounts
    )
)
