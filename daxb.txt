
let
    // Get the two existing queries
    Data = sheet1,
    HeadersTable = Headers,

    // Generate common temporary column names
    ColumnCount = List.Count(Table.ColumnNames(Data)),
    CommonNames = List.Transform(
        {1..ColumnCount},
        each "Column" & Text.From(_)
    ),

    // Check that both tables have the same number of columns
    CheckColumns =
        if ColumnCount <> List.Count(Table.ColumnNames(HeadersTable))
        then error "Column count mismatch between sheet1 and Headers"
        else true,

    // Convert the Headers column names into the first data row
    HeaderRow = Table.FromRows(
        {Table.ColumnNames(HeadersTable)},
        CommonNames
    ),

    // Rename data columns to common temporary names
    DataRenamed = Table.RenameColumns(
        Data,
        List.Zip({Table.ColumnNames(Data), CommonNames})
    ),

    // Append header row above actual data
    Combined = Table.Combine({HeaderRow, DataRenamed}),

    // Promote the first row as actual column headers
    FinalTable = Table.PromoteHeaders(
        Combined,
        [PromoteAllScalars = true]
    )

in
    FinalTable
