
Inactive Exited Accounts R12 =
SUMX(
    VALUES(
        'vw_factaccountfundmonth'[fundid]
    ),
    CALCULATE(
        MAX(
            'vw_factaccountfundmonth'
                [fund_fin_inactive_exited_accounts_r12]
        )
    )
)

---------------------------------------------------------------------------------------------------------

Total Exited Accounts R12 =
SUMX(
    VALUES(
        'vw_factaccountfundmonth'[fundid]
    ),
    CALCULATE(
        MAX(
            'vw_factaccountfundmonth'
                [fund_exited_accounts_r12]
        )
    )
)

-------------------------------------------------------------------------------------------


Inactive Exit Rate - End =
VAR _EndMonth =
    [Selected End Month]

VAR _MonthKey =
    YEAR(_EndMonth) * 100
        + MONTH(_EndMonth)

VAR _InactiveExited =
    CALCULATE(
        [Inactive Exited Accounts R12],
        REMOVEFILTERS('Calendar'),
        TREATAS(
            {_MonthKey},
            'vw_factaccountfundmonth'[monthkey]
        )
    )

VAR _TotalExited =
    CALCULATE(
        [Total Exited Accounts R12],
        REMOVEFILTERS('Calendar'),
        TREATAS(
            {_MonthKey},
            'vw_factaccountfundmonth'[monthkey]
        )
    )

RETURN
IF(
    ISBLANK(_EndMonth)
        || [Month Selection Valid] <> 1
        || ISBLANK(_TotalExited)
        || _TotalExited <= 0,
    BLANK(),
    DIVIDE(
        COALESCE(_InactiveExited, 0),
        _TotalExited
    )
)

Inactive Exit Rate - Change =
VAR _EndRate =
    [Inactive Exit Rate - End]

VAR _StartRate =
    [Inactive Exit Rate - Start]

RETURN
IF(
    ISBLANK(_EndRate)
        || ISBLANK(_StartRate),
    BLANK(),
    _EndRate - _StartRate
)

-------------------------------------------------------------------------------------------------------------


Inactive Exit Rate - Change Label =
VAR _Change =
    [Inactive Exit Rate - Change]

VAR _StartMonth =
    [Selected Start Month]

RETURN
IF(
    ISBLANK(_Change)
        || ISBLANK(_StartMonth),
    BLANK(),
    SWITCH(
        TRUE(),
        _Change > 0, "▲ ",
        _Change < 0, "▼ ",
        "● "
    )
    & FORMAT(ABS(_Change) * 100, "0.0")
    & "% vs "
    & FORMAT(_StartMonth, "MMM-yy")
)

--------------------------------------------------------------------------------------------------------------


Inactive Exit Rate - Change Color =
VAR _Change =
    [Inactive Exit Rate - Change]

RETURN
SWITCH(
    TRUE(),
    ISBLANK(_Change), "#808080",
    _Change > 0, "#008000",
    _Change < 0, "#D32F2F",
    "#808080"
)
