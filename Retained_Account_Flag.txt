Retained_Account_Flag =

VAR StartMonth =
    SELECTEDVALUE(Start_Month[MonthStart])

VAR EndMonth =
    SELECTEDVALUE(End_Month[MonthStart])

VAR CurrentAccount =
    SELECTEDVALUE(Query1[accountID])

VAR StartAccounts =
    CALCULATETABLE(
        VALUES(Query1[accountID]),
        REMOVEFILTERS(Query1[month_start]),
        Query1[month_start] = StartMonth,
        Query1[AccountStatus] = "Active"
    )

VAR EndAccounts =
    CALCULATETABLE(
        VALUES(Query1[accountID]),
        REMOVEFILTERS(Query1[month_start]),
        Query1[month_start] = EndMonth,
        Query1[AccountStatus] = "Active"
    )

RETURN
    IF(
        ISBLANK(StartMonth)
            || ISBLANK(EndMonth)
            || StartMonth >= EndMonth
            || ISBLANK(CurrentAccount),
        BLANK(),
        IF(
            CONTAINS(
                StartAccounts,
                Query1[accountID], CurrentAccount
            )
            &&
            CONTAINS(
                EndAccounts,
                Query1[accountID], CurrentAccount
            ),
            1,
            0
        )
    )
