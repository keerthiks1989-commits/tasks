
Retention Rate Accounts - Change Label =
VAR _Change =
    [Retention Rate Accounts - Change]

VAR _StartMonth =
    [Selected Start Month]

RETURN
IF(
    ISBLANK(_Change)
        || ISBLANK(_StartMonth),
    BLANK(),
    SWITCH(
        TRUE(),
        _Change > 0, "▲ ",
        _Change < 0, "▼ ",
        "● "
    )
    & FORMAT(ABS(_Change) * 100, "0.0")
    & " pp vs "
    & FORMAT(_StartMonth, "MMM-yy")
)
