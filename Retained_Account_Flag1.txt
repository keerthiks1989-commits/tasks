Retained_Account_Flag =

VAR StartMonth =
    SELECTEDVALUE(Start_Month[MonthStart])

VAR EndMonth =
    SELECTEDVALUE(End_Month[MonthStart])

VAR CurrentAccount =
    SELECTEDVALUE(Query1[accountID])

VAR StartCount =
    CALCULATE(
        COUNTROWS(Query1),
        REMOVEFILTERS(Query1),
        Query1[accountID] = CurrentAccount,
        Query1[month_start] = StartMonth,
        Query1[AccountStatus] = "Active"
    )

VAR EndCount =
    CALCULATE(
        COUNTROWS(Query1),
        REMOVEFILTERS(Query1),
        Query1[accountID] = CurrentAccount,
        Query1[month_start] = EndMonth,
        Query1[AccountStatus] = "Active"
    )

RETURN
    IF(
        ISBLANK(StartMonth)
            || ISBLANK(EndMonth)
            || StartMonth >= EndMonth
            || ISBLANK(CurrentAccount),
        BLANK(),
        IF(
            StartCount > 0 && EndCount > 0,
            1,
            0
        )
    )
