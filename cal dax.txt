
Calendar =
VAR _StartDate =
    DATE(2023, 4, 1)

VAR _LastMonth =
    MAXX(
        ALL('vw_factaccountfundmonth'),
        'vw_factaccountfundmonth'[monthyear]
    )

RETURN
ADDCOLUMNS(
    CALENDAR(
        _StartDate,
        EOMONTH(_LastMonth, 0)
    ),
    "MonthStart",
        DATE(YEAR([Date]), MONTH([Date]), 1),
    "MonthEnd",
        EOMONTH([Date], 0),
    "Monthyear",
        FORMAT([Date], "MMM-yy"),
    "Year",
        YEAR([Date]),
    "MonthNumber",
        MONTH([Date])
)


---------------------------------------------------------------------------------


Start_Month =
VAR _AvailableMonths =
    DISTINCT(
        SELECTCOLUMNS(
            FILTER(
                'vw_factaccountfundmonth',
                NOT ISBLANK(
                    'vw_factaccountfundmonth'[monthyear]
                )
            ),
            "MonthStart",
                DATE(
                    YEAR('vw_factaccountfundmonth'[monthyear]),
                    MONTH('vw_factaccountfundmonth'[monthyear]),
                    1
                )
        )
    )
RETURN
ADDCOLUMNS(
    _AvailableMonths,
    "MonthYear",
        FORMAT([MonthStart], "MMM-yy")
)
