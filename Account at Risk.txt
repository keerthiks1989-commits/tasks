
Accounts at Risk - End =
VAR _EndMonth =
    [Selected End Month]

VAR _MonthKey =
    YEAR(_EndMonth) * 100
        + MONTH(_EndMonth)

RETURN
IF(
    ISBLANK(_EndMonth)
        || [Month Selection Valid] <> 1,
    BLANK(),
    CALCULATE(
        DISTINCTCOUNT(
            'vw_factaccountfundmonth'[accountid]
        ),
        REMOVEFILTERS('Calendar'),
        TREATAS(
            {_MonthKey},
            'vw_factaccountfundmonth'[monthkey]
        ),
        KEEPFILTERS(
            'vw_factaccountfundmonth'[retention_category]
                = "At Risk"
        )
    )
)

------------------------------------------------------------------------------------------

Total Accounts - End =
VAR _EndMonth =
    [Selected End Month]

VAR _MonthKey =
    YEAR(_EndMonth) * 100
        + MONTH(_EndMonth)

RETURN
IF(
    ISBLANK(_EndMonth)
        || [Month Selection Valid] <> 1,
    BLANK(),
    CALCULATE(
        DISTINCTCOUNT(
            'vw_factaccountfundmonth'[accountid]
        ),
        REMOVEFILTERS('Calendar'),
        REMOVEFILTERS(
            'vw_factaccountfundmonth'[retention_category]
        ),
        TREATAS(
            {_MonthKey},
            'vw_factaccountfundmonth'[monthkey]
        )
    )
)

-------------------------------------------------------------------------------------

Accounts at Risk % - End =
DIVIDE(
    [Accounts at Risk - End],
    [Total Accounts - End]
)
----------------------------------------------------------------------------------------

Accounts at Risk Subtitle =
VAR _Percentage =
    [Accounts at Risk % - End]

RETURN
IF(
    ISBLANK(_Percentage),
    BLANK(),
    FORMAT(_Percentage, "0.0%")
        & " of Total Accounts"
)
